import { useState, useEffect } from "react";
import axios from "axios";

/**
 * Custom React hook for managing categories data
 * 
 * This hook fetches categories from the API and provides them to components.
 * It automatically fetches categories when the component mounts and handles
 * the loading state internally.
 * 
 * @returns {Array} categories - Array of category objects fetched from the API
 * 
 * @example
 * ```jsx
 * function CategoryList() {
 *   const categories = useCategory();
 *   
 *   return (
 *     <div>
 *       {categories.map(category => (
 *         <div key={category._id}>{category.name}</div>
 *       ))}
 *     </div>
 *   );
 * }
 * ```
 * 
 * @since 1.0.0
 * @author Your Name
 */
export default function useCategory() {
  /**
   * State to store the fetched categories
   * @type {Array}
   */
  const [categories, setCategories] = useState([]);

  /**
   * Fetches categories from the API endpoint
   * 
   * Makes a GET request to `/api/v1/category/get-category` and updates
   * the categories state with the response data. Handles errors by
   * logging them to the console.
   * 
   * @async
   * @function getCategories
   * @returns {Promise<void>}
   * 
   * @throws {Error} When the API request fails
   */
  const getCategories = async () => {
    try {
      const { data } = await axios.get("/api/v1/category/get-category");
      setCategories(data?.category);
    } catch (error) {
      console.log(error);
    }
  };

  /**
   * Effect hook that runs when the component mounts
   * 
   * Automatically fetches categories when the hook is first used.
   * The empty dependency array ensures this only runs once.
   */
  useEffect(() => {
    getCategories();
  }, []);

  return categories;
}
